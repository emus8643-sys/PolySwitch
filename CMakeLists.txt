cmake_minimum_required(VERSION 3.24)
project(polyswitch_native4 VERSION 0.4.18 LANGUAGES CXX)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
include(KDECMakeSettings)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick Quick3D ShaderTools DBus)
find_package(KF6 REQUIRED COMPONENTS CoreAddons GlobalAccel Config)
find_package(KWin REQUIRED)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
kcoreaddons_add_plugin(polyswitch_native4 SOURCES src/backgroundchooser.cpp src/effect.cpp src/prism.cpp src/facegeometry.cpp src/settingsstore.cpp INSTALL_NAMESPACE "kwin/effects/plugins")
target_include_directories(polyswitch_native4 PRIVATE src)
target_link_libraries(polyswitch_native4 PRIVATE Qt6::DBus KWin::kwin Qt6::Core Qt6::Gui Qt6::Quick Qt6::Quick3D KF6::CoreAddons KF6::GlobalAccel KF6::ConfigCore KF6::ConfigGui)
qt_add_resources(polyswitch_native4 qml_resources PREFIX "/polyswitch4" FILES qml/FirstRunSetup.qml qml/ProfileSettings.qml qml/PlasmaFrame.qml qml/PlasmaSvgFrame.qml qml/PlasmaButton.qml qml/EdgeLight.qml qml/CornerFlare.qml qml/FrameAtmosphere.qml qml/SunlightHalo.qml qml/SettingsComboBox.qml qml/GlowStroke.qml qml/main.qml qml/Switcher.qml qml/FacePreview.qml qml/HoverGlow.qml qml/SettingsPanel.qml qml/ColorSetting.qml qml/SliderSetting.qml qml/SettingsSwitch.qml qml/ScreenRotation.js)

qt_add_shaders(polyswitch_native4 surface_shaders PREFIX "/polyswitch4/qml/shaders" BASE "qml/shaders" FILES qml/shaders/edge.frag qml/shaders/flare.frag)

option(POLYSWITCH_BUILD_TESTS "Build isolated PolySwitch core tests" OFF)
if(POLYSWITCH_BUILD_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    enable_testing()
    add_executable(polyswitch_core_tests tests/test_core.cpp src/prism.cpp src/settingsstore.cpp)
    target_include_directories(polyswitch_core_tests PRIVATE src)
    target_compile_definitions(polyswitch_core_tests PRIVATE POLYSWITCH_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(polyswitch_core_tests PRIVATE Qt6::Quick Qt6::Quick3D Qt6::Test KF6::ConfigCore KF6::ConfigGui)
    add_executable(polyswitch_picker_tests tests/test_backgroundchooser.cpp src/backgroundchooser.cpp src/settingsstore.cpp)
    target_include_directories(polyswitch_picker_tests PRIVATE src)
    target_link_libraries(polyswitch_picker_tests PRIVATE Qt6::Gui Qt6::DBus Qt6::Test KF6::ConfigCore KF6::ConfigGui)
    find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
    add_test(NAME polyswitch_picker COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:polyswitch_picker_tests>)
    set_tests_properties(polyswitch_picker PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_test(NAME polyswitch_core COMMAND polyswitch_core_tests)
    set_tests_properties(polyswitch_core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
